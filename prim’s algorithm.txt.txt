#include <iostream>
#include <vector>
#include <climits>
using namespace std;

int main() {
    int n;
    cout << "Enter number of vertices: ";
    cin >> n;

    if (n <= 0) return 0;

    vector<vector<int>> a(n, vector<int>(n));
    vector<int> key(n, INT_MAX), parent(n, -1);
    vector<bool> vis(n, false);

    cout << "Enter adjacency matrix:\n";
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> a[i][j];

    key[0] = 0;

    for (int i = 0; i < n; i++) {
        int u = -1;

        for (int j = 0; j < n; j++)
            if (!vis[j] && (u == -1 || key[j] < key[u]))
                u = j;

        if (u == -1 || key[u] == INT_MAX) {
            cout << "MST not possible";
            return 0;
        }

        vis[u] = true;

        for (int v = 0; v < n; v++)
            if (a[u][v] > 0 && !vis[v] && a[u][v] < key[v]) {
                key[v] = a[u][v];
                parent[v] = u;
            }
    }

    int total = 0;
    cout << "Edges in MST:\n";

    for (int i = 1; i < n; i++) {
        cout << parent[i] << " - " << i << " : " << key[i] << '\n';
        total += key[i];
    }

    cout << "Minimum cost: " << total << '\n';
    return 0;
}
